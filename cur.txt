main.py — FastAPI Application Entry Point

Exposes three surfaces:
  1. POST /webhook/github     — GitHub Webhook receiver (triggers review graph)
  2. POST /hitl/respond       — HITL decision endpoint (resumes paused graph)
  3. GET  /review/{thread_id} — Poll review status and results
#   4. GET  /health             — Health check

Webhook Security:
  GitHub HMAC-SHA256 signature is verified before any processing.
"""

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Warm up the vector store on startup (avoids cold-start latency)."""
    log.info("server_starting")
    try:
        get_knowledge_base()   # preload vector store into memory
        log.info("knowledge_base_ready")
    except Exception as exc:
        log.error("knowledge_base_init_failed", error=str(exc))
    yield
    log.info("server_shutting_down")

class WebhookPayload(BaseModel):
    """Parsed GitHub PR webhook body (only fields we care about)."""
    action: str
    pull_request: Dict[str, Any]
    repository: Dict[str, Any]


class HITLResponse(BaseModel):
    """Body sent by human reviewer to the /hitl/respond endpoint."""
    pr_url: str
    thread_id: str
    reviewer: str
    decision: Literal["approve", "request_changes", "escalate"]
    comment: str

    @field_validator("decision")
    @classmethod
    def validate_decision(cls, v: str) -> str:
        valid = {"approve", "request_changes", "escalate"}
        if v not in valid:
            raise ValueError(f"decision must be one of {valid}")
        return v


class ReviewTriggerRequest(BaseModel):
    """Direct API trigger (for testing without a GitHub webhook)."""
    pr_url: str


# ══════════════════════════════════════════════════════════════════════════════
# Webhook Signature Verification
# ══════════════════════════════════════════════════════════════════════════════

def _verify_github_signature(
    payload_bytes: bytes,
    signature_header: Optional[str],
    secret: str,
) -> bool:
    """Verify GitHub's HMAC-SHA256 webhook signature."""
    if not signature_header or not signature_header.startswith("sha256="):
        return False
    expected = hmac.new(
        secret.encode("utf-8"),
        payload_bytes,
        hashlib.sha256,
    ).hexdigest()
    received = signature_header.removeprefix("sha256=")
    return hmac.compare_digest(expected, received)


# ══════════════════════════════════════════════════════════════════════════════
# Background Task
# ══════════════════════════════════════════════════════════════════════════════

async def _run_review_graph(pr_url: str, thread_id: str) -> None:
    """
    Execute the full multi-agent review graph in a background task.
    Each run gets its own `thread_id` for LangGraph checkpointing.
    """
    log.info("graph_execution_started", pr_url=pr_url, thread_id=thread_id)
    try:
        config = {"configurable": {"thread_id": thread_id}}
        state = initial_state(pr_url)
        result = await pr_review_graph.ainvoke(state, config=config)

        if result.get("awaiting_human"):
            log.info("graph_interrupted_awaiting_human", thread_id=thread_id)
        else:
            log.info(
                "graph_completed",
                thread_id=thread_id,
                merge_ready=result.get("is_ready_for_merge"),
                comment_id=result.get("github_comment_id"),
            )
    except Exception as exc:
        log.error("graph_execution_failed", error=str(exc), thread_id=thread_id)

@app.post("/webhook/github", tags=["Webhook"], status_code=status.HTTP_202_ACCEPTED)
async def github_webhook(
    request: Request,
    background_tasks: BackgroundTasks,
    settings: Settings = Depends(get_settings),
) -> dict:
    """
    Receive GitHub PR webhook events.

    GitHub sends a POST request with HMAC-SHA256 signature whenever a PR
    is opened, synchronised (new commit pushed), or reopened.
    """
    payload_bytes = await request.body()

    # ── Security: verify webhook signature ───────────────────────────────────
    signature = request.headers.get("X-Hub-Signature-256")
    if not _verify_github_signature(payload_bytes, signature, settings.github_webhook_secret):
        log.warning("webhook_signature_invalid", ip=request.client.host if request.client else "unknown")
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid webhook signature.",
        )

    # ── Parse event ───────────────────────────────────────────────────────────
    event_type = request.headers.get("X-GitHub-Event", "")
    if event_type != "pull_request":
        return {"status": "ignored", "reason": f"event type '{event_type}' not handled"}

    payload = json.loads(payload_bytes)
    action = payload.get("action", "")
    if action not in ("opened", "synchronize", "reopened"):
        return {"status": "ignored", "reason": f"action '{action}' not handled"}

    # ── Extract PR URL ────────────────────────────────────────────────────────
    pr_html_url = payload.get("pull_request", {}).get("html_url")
    if not pr_html_url:
        raise HTTPException(status_code=400, detail="Missing pull_request.html_url")

    thread_id = f"pr-{uuid.uuid4().hex[:12]}"
    log.info("webhook_accepted", pr_url=pr_html_url, action=action, thread_id=thread_id)

    # ── Dispatch graph asynchronously ─────────────────────────────────────────
    background_tasks.add_task(_run_review_graph, pr_html_url, thread_id)

    return {"status": "accepted", "thread_id": thread_id, "pr_url": pr_html_url}


@app.post("/review/trigger", tags=["Review"], status_code=status.HTTP_202_ACCEPTED)
async def trigger_review_manually(
    body: ReviewTriggerRequest,
    background_tasks: BackgroundTasks,
) -> dict:
    """
    Manually trigger a PR review without a webhook.
    Useful for testing and CI integrations.
    """
    thread_id = f"manual-{uuid.uuid4().hex[:12]}"
    log.info("manual_review_triggered", pr_url=body.pr_url, thread_id=thread_id)
    background_tasks.add_task(_run_review_graph, body.pr_url, thread_id)
    return {"status": "accepted", "thread_id": thread_id}


@app.post("/hitl/respond", tags=["HITL"])
async def hitl_respond(
    body: HITLResponse,
    background_tasks: BackgroundTasks,
) -> dict:
    """
    Accept a human reviewer's decision and resume the paused graph.

    The thread_id links this decision to the interrupted graph run.
    """
    log.info(
        "hitl_response_received",
        thread_id=body.thread_id,
        decision=body.decision,
        reviewer=body.reviewer,
    )

    from datetime import datetime, timezone

    hitl_decision = HITLDecision(
        reviewer=body.reviewer,
        decision=body.decision,
        comment=body.comment,
        timestamp=datetime.now(timezone.utc).isoformat(),
    )

    async def _resume(thread_id: str, pr_url: str, decision: HITLDecision) -> None:
        """Inject the decision into state and resume the interrupted graph."""
        try:
            config = {"configurable": {"thread_id": thread_id}}
            # Fetch current state, inject decision, resume
            current_state = pr_review_graph.get_state(config)
            updated_values = {**current_state.values, "hitl_decision": decision}
            result = await pr_review_graph.ainvoke(updated_values, config=config)
            log.info(
                "graph_resumed_and_completed",
                thread_id=thread_id,
                decision=decision["decision"],
                merge_ready=result.get("is_ready_for_merge"),
            )
        except Exception as exc:
            log.error("graph_resume_failed", error=str(exc), thread_id=body.thread_id)

    background_tasks.add_task(_resume, body.thread_id, body.pr_url, hitl_decision)

    return {
        "status": "decision_received",
        "thread_id": body.thread_id,
        "decision": body.decision,
        "reviewer": body.reviewer,
    }


@app.get("/review/{thread_id}", tags=["Review"])
async def get_review_status(thread_id: str) -> dict:
    """
    Poll the status and results of a running or completed review.
    The thread_id is returned when the review is triggered.
    """
    try:
        config = {"configurable": {"thread_id": thread_id}}
        state = pr_review_graph.get_state(config)
        if not state or not state.values:
            raise HTTPException(status_code=404, detail=f"No review found for thread_id={thread_id}")

        values = state.values
        return {
            "thread_id": thread_id,
            "pr_url": values.get("pr_url"),
            "status": "awaiting_human" if values.get("awaiting_human") else "completed",
            "is_ready_for_merge": values.get("is_ready_for_merge"),
            "security_score": values.get("security_score"),
            "hitl_required": values.get("hitl_required"),
            "github_comment_id": values.get("github_comment_id"),
            "error": values.get("error"),
            "execution_log": values.get("node_execution_log", []),
        }
    except HTTPException:
        raise
    except Exception as exc:
        log.error("status_fetch_failed", error=str(exc))
        raise HTTPException(status_code=500, detail=str(exc))
